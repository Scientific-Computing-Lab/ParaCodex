/* maximum power density possible (say 300W for a 10mm x 10mm chip) */
#define MAX_PD (3.0e6)
/* required precision in degrees */
#define PRECISION 0.001
#define SPEC_HEAT_SI 1.75e6
#define K_SI 100
/* capacitance fitting factor */
#define FACTOR_CHIP 0.5

typedef float FLOAT;

/* chip parameters */
const FLOAT t_chip = 0.0005;
const FLOAT chip_height = 0.016;
const FLOAT chip_width = 0.016;

/* ambient temperature, assuming no package at all */
const FLOAT amb_temp = 80.0;

int num_omp_threads;

/* Single iteration of the transient solver in the grid model.
 * advances the solution of the discretized difference equations
 * by one time step
 */
void single_iteration(FLOAT *result, FLOAT *temp, FLOAT *power, int row, int col,
                      FLOAT Cap_1, FLOAT Rx_1, FLOAT Ry_1, FLOAT Rz_1,
                      FLOAT step)
{
    (void)step;
    int total = row * col;

#pragma omp target teams distribute parallel for collapse(2) default(none) \
    map(to : temp[0:total], power[0:total]) map(from : result[0:total]) \
    shared(result, temp, power, row, col, Cap_1, Rx_1, Ry_1, Rz_1)
    for (int r = 0; r < row; ++r) {
        for (int c = 0; c < col; ++c) {
            int idx = r * col + c;
            FLOAT cur = temp[idx];
            FLOAT delta = power[idx];

            if (r > 0) {
                delta += (temp[(r - 1) * col + c] - cur) * Ry_1;
            }
            if (r + 1 < row) {
                delta += (temp[(r + 1) * col + c] - cur) * Ry_1;
            }
            if (c > 0) {
                delta += (temp[r * col + c - 1] - cur) * Rx_1;
            }
            if (c + 1 < col) {
                delta += (temp[r * col + c + 1] - cur) * Rx_1;
            }

            result[idx] = cur + Cap_1 * (delta + (amb_temp - cur) * Rz_1);
        }
    }
}

/* Transient solver driver routine: simply converts the heat
 * transfer differential equations to difference equations
 * and solves the difference equations by iterating
 */
void compute_tran_temp(FLOAT *result, int num_iterations, FLOAT *temp, FLOAT *power, int row, int col)
{
#ifdef VERBOSE
    int i = 0;
#endif

    FLOAT grid_height = chip_height / row;
    FLOAT grid_width = chip_width / col;

    FLOAT Cap = FACTOR_CHIP * SPEC_HEAT_SI * t_chip * grid_width * grid_height;
    FLOAT Rx = grid_width / (2.0 * K_SI * t_chip * grid_height);
    FLOAT Ry = grid_height / (2.0 * K_SI * t_chip * grid_width);
    FLOAT Rz = t_chip / (K_SI * grid_height * grid_width);

    FLOAT max_slope = MAX_PD / (FACTOR_CHIP * t_chip * SPEC_HEAT_SI);
    FLOAT step = PRECISION / max_slope / 1000.0;

    FLOAT Rx_1 = 1.f / Rx;
    FLOAT Ry_1 = 1.f / Ry;
    FLOAT Rz_1 = 1.f / Rz;
    FLOAT Cap_1 = step / Cap;
#ifdef VERBOSE
    fprintf(stdout, "total iterations: %d s\tstep size: %g s\n", num_iterations, step);
    fprintf(stdout, "Rx: %g\tRy: %g\tRz: %g\tCap: %g\n", Rx, Ry, Rz, Cap);
#endif

    {
        FLOAT *r = result;
        FLOAT *t = temp;
        for (int i = 0; i < num_iterations; i++) {
#ifdef VERBOSE
            fprintf(stdout, "iteration %d\n", i++);
#endif
            single_iteration(r, t, power, row, col, Cap_1, Rx_1, Ry_1, Rz_1, step);
            FLOAT *tmp = t;
            t = r;
            r = tmp;
        }
    }
#ifdef VERBOSE
    fprintf(stdout, "iteration %d\n", i++);
#endif
}

void single_iteration(FLOAT *result, FLOAT *temp, FLOAT *power, int row, int col,
					  FLOAT Cap_1, FLOAT Rx_1, FLOAT Ry_1, FLOAT Rz_1,
					  FLOAT step)
{
    FLOAT delta;
    int r, c;
    int chunk;
    int num_chunk = row*col / (BLOCK_SIZE_R * BLOCK_SIZE_C);
    int chunks_in_row = col/BLOCK_SIZE_C;
    int chunks_in_col = row/BLOCK_SIZE_R;

    for ( chunk = 0; chunk < num_chunk; ++chunk )
    {
        int r_start = BLOCK_SIZE_R*(chunk/chunks_in_col);
        int c_start = BLOCK_SIZE_C*(chunk%chunks_in_row);
        int r_end = r_start + BLOCK_SIZE_R > row ? row : r_start + BLOCK_SIZE_R;
        int c_end = c_start + BLOCK_SIZE_C > col ? col : c_start + BLOCK_SIZE_C;

        if ( r_start == 0 || c_start == 0 || r_end == row || c_end == col )
        {
            for ( r = r_start; r < r_start + BLOCK_SIZE_R; ++r ) {
                for ( c = c_start; c < c_start + BLOCK_SIZE_C; ++c ) {
                    /* Corner 1 */
                    if ( (r == 0) && (c == 0) ) {
                        delta = (Cap_1) * (power[0] +
                            (temp[1] - temp[0]) * Rx_1 +
                            (temp[col] - temp[0]) * Ry_1 +
                            (amb_temp - temp[0]) * Rz_1);
                    }	/* Corner 2 */
                    else if ((r == 0) && (c == col-1)) {
                        delta = (Cap_1) * (power[c] +
                            (temp[c-1] - temp[c]) * Rx_1 +
                            (temp[c+col] - temp[c]) * Ry_1 +
                        (   amb_temp - temp[c]) * Rz_1);
                    }	/* Corner 3 */
                    else if ((r == row-1) && (c == col-1)) {
                        delta = (Cap_1) * (power[r*col+c] +
                            (temp[r*col+c-1] - temp[r*col+c]) * Rx_1 +
                            (temp[(r-1)*col+c] - temp[r*col+c]) * Ry_1 +
                        (   amb_temp - temp[r*col+c]) * Rz_1);
                    }	/* Corner 4	*/
                    else if ((r == row-1) && (c == 0)) {
                        delta = (Cap_1) * (power[r*col] +
                            (temp[r*col+1] - temp[r*col]) * Rx_1 +

                            (temp[c+1] + temp[c-1] - 2.0*temp[c]) * Rx_1 +

/* Single iteration of the transient solver in the grid model.
 * advances the solution of the discretized difference equations
 * by one time step
 */
static inline void single_iteration(FLOAT *__restrict__ result, const FLOAT *__restrict__ temp, const FLOAT *__restrict__ power,
                                    int row, int col, FLOAT Cap_1, FLOAT Rx_1, FLOAT Ry_1, FLOAT Rz_1)
{
    const int last_row = row - 1;
    const int last_col = col - 1;

    #pragma omp target teams loop collapse(2) is_device_ptr(result, temp, power)
    for (int r = 0; r < row; ++r)
    {
        const int row_offset = r * col;
        const int row_above = row_offset - col;
        const int row_below = row_offset + col;

        for (int c = 0; c < col; ++c)
        {
            const int idx = row_offset + c;
            const FLOAT center = temp[idx];
            FLOAT delta;

            if (r > 0 && r < last_row && c > 0 && c < last_col)
            {
                const FLOAT temp_north = temp[row_above + c];
                const FLOAT temp_south = temp[row_below + c];
                const FLOAT temp_west = temp[idx - 1];
                const FLOAT temp_east = temp[idx + 1];
                delta = Cap_1 * (power[idx] +
                    (temp_south + temp_north - 2.0f * center) * Ry_1 +
                    (temp_east + temp_west - 2.0f * center) * Rx_1 +
                    (amb_temp - center) * Rz_1);
            }
            else if ((r == 0) && (c == 0))
            {
                delta = Cap_1 * (power[0] +
                    (temp[1] - center) * Rx_1 +
                    (temp[col] - center) * Ry_1 +
                    (amb_temp - center) * Rz_1);
            }
            else if ((r == 0) && (c == last_col))
            {
                delta = Cap_1 * (power[c] +
                    (temp[c - 1] - center) * Rx_1 +
                    (temp[row_offset + col] - center) * Ry_1 +
                    (amb_temp - center) * Rz_1);
            }
            else if ((r == last_row) && (c == last_col))
            {
                delta = Cap_1 * (power[idx] +
                    (temp[idx - 1] - center) * Rx_1 +
                    (temp[row_above + c] - center) * Ry_1 +
                    (amb_temp - center) * Rz_1);
            }
            else if ((r == last_row) && (c == 0))
            {
                delta = Cap_1 * (power[row_offset] +
                    (temp[row_offset + 1] - center) * Rx_1 +
                    (temp[row_above] - center) * Ry_1 +
                    (amb_temp - center) * Rz_1);
            }
            else if (r == 0)
            {
                delta = Cap_1 * (power[c] +
                    (temp[c + 1] + temp[c - 1] - 2.0f * center) * Rx_1 +
                    (temp[col + c] - center) * Ry_1 +
                    (amb_temp - center) * Rz_1);
            }
            else if (c == last_col)
            {
                delta = Cap_1 * (power[idx] +
                    (temp[row_below + c] + temp[row_above + c] - 2.0f * center) * Ry_1 +
                    (temp[idx - 1] - center) * Rx_1 +
                    (amb_temp - center) * Rz_1);
            }
            else if (r == last_row)
            {
                delta = Cap_1 * (power[idx] +
                    (temp[idx + 1] + temp[idx - 1] - 2.0f * center) * Rx_1 +

                    (temp[idx + 1] - center) * Rx_1 +
                    (amb_temp - center) * Rz_1);
            }

            result[idx] = center + delta;
        }
    }
}

/* Transient solver driver routine: simply converts the heat
 * transfer differential equations to difference equations
 * and solves the difference equations by iterating
 */
void compute_tran_temp(FLOAT *result, int num_iterations, FLOAT *temp, FLOAT *power, int row, int col)
{
	#ifdef VERBOSE
	int i = 0;
	#endif

	FLOAT grid_height = chip_height / row;
	FLOAT grid_width = chip_width / col;

	FLOAT Cap = FACTOR_CHIP * SPEC_HEAT_SI * t_chip * grid_width * grid_height;
	FLOAT Rx = grid_width / (2.0 * K_SI * t_chip * grid_height);
	FLOAT Ry = grid_height / (2.0 * K_SI * t_chip * grid_width);
	FLOAT Rz = t_chip / (K_SI * grid_height * grid_width);

	FLOAT max_slope = MAX_PD / (FACTOR_CHIP * t_chip * SPEC_HEAT_SI);
    FLOAT step = PRECISION / max_slope / 1000.0;

    FLOAT Rx_1=1.f/Rx;
    FLOAT Ry_1=1.f/Ry;
    FLOAT Rz_1=1.f/Rz;
    FLOAT Cap_1 = step/Cap;
    const size_t bytes = (size_t)row * (size_t)col * sizeof(FLOAT);
    const int host_device = omp_get_initial_device();
    const int device = omp_get_default_device();
    FLOAT *d_result = (FLOAT *) omp_target_alloc(bytes, device);
    FLOAT *d_temp = (FLOAT *) omp_target_alloc(bytes, device);
    FLOAT *d_power = (FLOAT *) omp_target_alloc(bytes, device);

    if (!d_result || !d_temp || !d_power) {
        fatal("unable to allocate device memory");
    }

    omp_target_memcpy(d_temp, temp, bytes, 0, 0, device, host_device);
    omp_target_memcpy(d_result, result, bytes, 0, 0, device, host_device);
    omp_target_memcpy(d_power, power, bytes, 0, 0, device, host_device);

	#ifdef VERBOSE
	fprintf(stdout, "total iterations: %d s\tstep size: %g s\n", num_iterations, step);
	fprintf(stdout, "Rx: %g\tRy: %g\tRz: %g\tCap: %g\n", Rx, Ry, Rz, Cap);
	#endif

    {
        FLOAT *d_result_iter = d_result;
        FLOAT *d_temp_iter = d_temp;

        for (int i = 0; i < num_iterations; i++)
        {
            #ifdef VERBOSE
            fprintf(stdout, "iteration %d\n", i++);
            #endif
            single_iteration(d_result_iter, d_temp_iter, d_power, row, col, Cap_1, Rx_1, Ry_1, Rz_1);
            FLOAT *tmp = d_temp_iter;
            d_temp_iter = d_result_iter;
            d_result_iter = tmp;
        }

        omp_target_memcpy(temp, d_temp, bytes, 0, 0, host_device, device);
        omp_target_memcpy(result, d_result, bytes, 0, 0, host_device, device);
    }

    omp_target_free(d_power, device);
    omp_target_free(d_temp, device);
    omp_target_free(d_result, device);
	#ifdef VERBOSE
	fprintf(stdout, "iteration %d\n", i++);
	#endif
}

void writeoutput(FLOAT *vect, int grid_rows, int grid_cols, char *file) {

    int i, j;
    FILE *fp;
    char str[STR_SIZE];

    if( (fp = fopen(file, "w" )) == 0 )
        printf( "The file was not opened\n" );


    for (i=0; i < grid_rows; i++)
        for (j=0; j < grid_cols; j++)
        {

            sprintf(str, "%d\t%g\n", i*grid_cols+j, vect[i*grid_cols+j]);
            fputs(str,fp);
        }

    fclose(fp);
}

void read_input(FLOAT *vect, int grid_rows, int grid_cols, char *file)
{
  	int i;
	FILE *fp;
	char str[STR_SIZE];
	FLOAT val;

	fp = fopen (file, "r");
	if (!fp)
		fatal ("file could not be opened for reading");

	for (i=0; i < grid_rows * grid_cols; i++) {
		fgets(str, STR_SIZE, fp);
		if (feof(fp))
			fatal("not enough lines in file");
		if ((sscanf(str, "%f", &val) != 1) )
			fatal("invalid file format");
		vect[i] = val;
	}

	fclose(fp);
}
